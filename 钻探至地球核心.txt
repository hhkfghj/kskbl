function HUB.new(class, props)
	local inst = Instance.new(class)
	local parent
	if props then
		for k, v in pairs(props) do
			if k == "Parent" then
				parent = v
			else
				inst[k] = v
			end
		end
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

function HUB.tween(inst, time, props, style, dir)
	local tween = TweenService:Create(
		inst,
		TweenInfo.new(time or 0.18, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	tween:Play()
	return tween
end

function HUB.connect(sig, fn)
	local conn = sig:Connect(fn)
	table.insert(HUB.Conns, conn)
	return conn
end

function HUB.cleanup(fn)
	table.insert(HUB.Cleanup, fn)
end

-- 取一个不会被游戏 UI 遮挡、也不被销毁的父容器
function HUB.container()
	if type(gethui) == "function" then
		local ok, hui = pcall(gethui)
		if ok and hui then
			return hui
		end
	end
	if type(hiddenUi) == "function" then
		local ok, hui = pcall(hiddenUi)
		if ok and hui then
			return hui
		end
	end
	local ok, cg = pcall(function()
		return game:GetService("CoreGui")
	end)
	if ok and cg then
		return cg
	end
	return LP:WaitForChild("PlayerGui")
end

function HUB.character()
	local char = LP.Character
	if not char then
		char = LP.CharacterAdded:Wait()
	end
	return char
end

function HUB.humanoid()
	local char = LP.Character
	if not char then
		return nil
	end
	return char:FindFirstChildOfClass("Humanoid")
end

function HUB.rootPart()
	local char = LP.Character
	if not char then
		return nil
	end
	return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso")
end

function HUB.isAlive()
	local hum = HUB.humanoid()
	return hum ~= nil and hum.Health > 0
end

function HUB.distanceTo(part)
	local root = HUB.rootPart()
	if not root or not part then
		return math.huge
	end
	return (root.Position - part.Position).Magnitude
end

function HUB.formatNum(n)
	n = tonumber(n) or 0
	if n >= 1e9 then
		return string.format("%.2fB", n / 1e9)
	elseif n >= 1e6 then
		return string.format("%.2fM", n / 1e6)
	elseif n >= 1e3 then
		return string.format("%.1fK", n / 1e3)
	end
	return tostring(math.floor(n))
end

-- 世界坐标 -> 屏幕坐标
function HUB.worldToScreen(pos)
	local cam = Workspace.CurrentCamera
	if not cam then
		return nil
	end
	local sp, onScreen = cam:WorldToViewportPoint(pos)
	if not onScreen then
		return nil
	end
	return Vector2.new(sp.X, sp.Y), sp.Z
end

-- 触发一个 clickdetector / proximityprompt（自动化用）
function HUB.firePrompt(part, distance)
	local prompt = part:FindFirstChildWhichIsA("ProximityPrompt", true)
	if prompt and type(fireproximityprompt) == "function" then
		pcall(fireproximityprompt, prompt)
		return true
	end
	local detector = part:FindFirstChildWhichIsA("ClickDetector", true)
	if detector and type(fireclickdetector) == "function" then
		pcall(fireclickdetector, detector)
		return true
	end
	return false
end

function HUB.isAllyModel(model)
	if not model then
		return true
	end
	if HUB.FriendlyNames[model.Name] then
		return true
	end
	-- 玩家角色
	local plr = Players:GetPlayerFromCharacter(model)
	if plr then
		return true
	end
	-- 有 OwnerPlayer / 已招募的跟班
	if model:GetAttribute("OwnerUserId") then
		return true
	end
	return false
end

function HUB.isEnemyModel(model)
	if not model or HUB.isAllyModel(model) then
		return false
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then
		return false
	end
	if HUB.HostileNames[model.Name] then
		return true
	end
	-- 兜底：Npc 文件夹里的活物都算怪
	if model.Parent and model.Parent.Name == "Npc" then
		return true
	end
	return false
end

local function rootOf(model)
	return model:FindFirstChild("HumanoidRootPart")
		or model:FindFirstChild("UpperTorso")
		or model:FindFirstChildWhichIsA("BasePart", true)
end

-- 全部敌对单位，按距离排序
function HUB.getEnemies(maxDist)
	local out = {}
	local myRoot = HUB.rootPart()
	if not myRoot then
		return out
	end
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("Humanoid") and inst.Parent and inst.Health > 0 then
			local model = inst.Parent
			if HUB.isEnemyModel(model) then
				local part = rootOf(model)
				if part then
					local dist = (part.Position - myRoot.Position).Magnitude
					if not maxDist or dist <= maxDist then
						table.insert(out, { model = model, part = part, hum = inst, dist = dist })
					end
				end
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 最近的敌对单位
function HUB.getClosestEnemy(maxDist)
	local list = HUB.getEnemies(maxDist)
	return list[1]
end

-- 全部矿石节点（Part 上带 SessionOreId）
function HUB.getOres(maxDist, typeFilter)
	local out = {}
	local myRoot = HUB.rootPart()
	if not myRoot then
		return out
	end
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("BasePart") and inst:GetAttribute("SessionOreId") then
			if inst.Parent then
				local oreType = inst:GetAttribute("SessionOreType")
				if not typeFilter or typeFilter == "全部" or tostring(oreType) == typeFilter then
					local dist = (inst.Position - myRoot.Position).Magnitude
					if not maxDist or dist <= maxDist then
						table.insert(out, { part = inst, dist = dist, oreType = oreType, node = inst })
					end
				end
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 地上的掉落物
function HUB.getDroppedItems(maxDist)
	local out = {}
	local myRoot = HUB.rootPart()
	local folder = Workspace:FindFirstChild("Items")
	if not folder or not myRoot then
		return out
	end
	for _, item in ipairs(folder:GetChildren()) do
		-- 商店展示件（带 ShopItem 标签）不能捡，直接跳过
		if not CollectionService:HasTag(item, "ShopItem") then
			local part
			if item:IsA("BasePart") then
				part = item
			elseif item:IsA("Model") then
				part = item:FindFirstChild("Root") or item:FindFirstChildWhichIsA("BasePart", true)
			end
			if part then
				local dist = (part.Position - myRoot.Position).Magnitude
				if not maxDist or dist <= maxDist then
					table.insert(out, { model = item, part = part, dist = dist, name = item.Name })
				end
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 宝箱：只认世界里真正能交互的箱子（ChestNode 下、带 ProximityPrompt）
-- 地上掉的护甲（Steel_Chest / Rusty_Chest）和 Props 里的 CrateGroup/ChestStep 都不算
function HUB.getChests(maxDist)
	local out = {}
	local myRoot = HUB.rootPart()
	if not myRoot then
		return out
	end
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("Model") then
			local lower = string.lower(inst.Name)
			if string.find(lower, "chest") or string.find(lower, "crate") then
				local skip = false
				-- 地上的掉落物（护甲、道具）不算宝箱
				local parentName = inst.Parent and inst.Parent.Name or ""
				if parentName == "Items" then
					skip = true
				end
				local def = HUB.getItemDef and HUB.getItemDef(inst.Name)
				if def and def.Attributes and def.Attributes.Category == "Armor" then
					skip = true
				end
				-- Props 里的装饰物（CrateGroup / ChestStep）不算
				if string.find(lower, "crate") or string.find(lower, "step") then
					skip = true
				end
				if not skip then
					local hasPrompt = false
					for _, d in ipairs(inst:GetDescendants()) do
						if d:IsA("ProximityPrompt") then
							hasPrompt = true
							break
						end
					end
					local isNode = string.find(parentName, "ChestNode") ~= nil or string.find(parentName, "Chests") ~= nil
					if hasPrompt and isNode then
						local part = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)
						if part then
							local dist = (part.Position - myRoot.Position).Magnitude
							if not maxDist or dist <= maxDist then
								table.insert(out, { model = inst, part = part, dist = dist, name = inst.Name })
							end
						end
					end
				end
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 地牢门
function HUB.getGates()
	local out = {}
	for _, inst in ipairs(Workspace:GetChildren()) do
		if inst:IsA("Model") and string.find(inst.Name, "DungeonGate") then
			local part = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart", true)
			if part then
				local prompt
				for _, d in ipairs(inst:GetDescendants()) do
					if d:IsA("ProximityPrompt") and string.find(string.lower(tostring(d.ObjectText)), "dungeon") then
						prompt = d
					end
				end
				local tier = string.match(inst.Name, "Tier(%d)")
				table.insert(out, {
					model = inst,
					part = part,
					tier = tier and tonumber(tier) or 0,
					prompt = prompt,
					dist = HUB.distanceTo(part),
				})
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 全部血量很高的单位（BOSS）
function HUB.getBosses(minHealth)
	minHealth = minHealth or 2000
	local out = {}
	local myRoot = HUB.rootPart()
	if not myRoot then
		return out
	end
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("Humanoid") and inst.Parent and inst.MaxHealth >= minHealth and inst.Health > 0 then
			local part = rootOf(inst.Parent)
			if part then
				table.insert(out, {
					model = inst.Parent,
					part = part,
					hum = inst,
					dist = (part.Position - myRoot.Position).Magnitude,
				})
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 物品数据库（354 件，含游戏自己的分类/价值/稀有度）
local itemDefs = nil
function HUB.getItemDefs()
	if itemDefs then
		return itemDefs
	end
	local ok, defs = pcall(function()
		local rs = game:GetService("ReplicatedStorage")
		local shared = rs:FindFirstChild("Shared")
		local db = shared and shared:FindFirstChild("ItemDatabase")
		local defsMod = db and db:FindFirstChild("Definitions")
		if not defsMod then
			return nil
		end
		return require(defsMod)
	end)
	itemDefs = (ok and type(defs) == "table") and defs or {}
	return itemDefs
end

function HUB.getItemDef(id)
	if not id then
		return nil
	end
	return HUB.getItemDefs()[id]
end

-- 游戏分类 -> 界面分类
local CATEGORY_MAP = {
	["Melee"] = "武器",
	["Primary Gun"] = "武器",
	["Secondary Gun"] = "武器",
	["Launcher"] = "武器",
	["Pickaxe"] = "镐子",
	["Ammo"] = "子弹",
	["Armor"] = "装备",
	["Scrap"] = "卖钱",
	["Corpse"] = "尸体",
}

function HUB.itemCategory(name)
	local id = tostring(name)
	local def = HUB.getItemDef(id)
	local gameCat = def and def.Attributes and def.Attributes.Category
	local n = string.lower(id)
	if gameCat then
		if gameCat == "Consumable" then
			if string.find(n, "potion") or string.find(n, "elixir") then
				return "药水"
			end
			return "医疗包"
		end
		local mapped = CATEGORY_MAP[gameCat]
		if mapped then
			return mapped
		end
		return "其他"
	end
	-- 数据库里没有的，按名字兜底
	if string.find(n, "pickaxe") then
		return "镐子"
	end
	if string.find(n, "ammo") then
		return "子弹"
	end
	if string.find(n, "medkit") or string.find(n, "bandage") then
		return "医疗包"
	end
	if string.find(n, "potion") then
		return "药水"
	end
	if string.find(n, "sword") or string.find(n, "rifle") or string.find(n, "revolver")
		or string.find(n, "shotgun") or string.find(n, "launcher") or string.find(n, "hammer") then
		return "武器"
	end
	if string.find(n, "boot") or string.find(n, "helmet") or string.find(n, "legs")
		or string.find(n, "chest") or string.find(n, "armor") then
		return "装备"
	end
	return "其他"
end

HUB.ItemCategories = { "武器", "镐子", "医疗包", "药水", "子弹", "装备", "卖钱", "尸体", "其他" }

-- 整张物品表里，某一大类的全部物品（中文名），不依赖地上有没有
function HUB.allItemNamesByCat(cat)
	local seen, out = {}, {}
	for id in pairs(HUB.getItemDefs()) do
		if HUB.itemCategory(id) == cat then
			local zh = HUB.zhName(id)
			if not seen[zh] then
				seen[zh] = true
				out[#out + 1] = zh
			end
		end
	end
	table.sort(out)
	return out
end

function HUB.chestTier(name)
	local n = string.lower(tostring(name))
	if string.find(n, "gold") or string.find(n, "diamond") or string.find(n, "void")
		or string.find(n, "core") or string.find(n, "phoenix") or string.find(n, "royal")
		or string.find(n, "legend") or string.find(n, "epic") then
		return "高级"
	end
	return "普通"
end


-- 物品价值（卖钱用）
function HUB.itemValue(item)
	local v = item:GetAttribute("Value")
	if v == nil then
		v = item:GetAttribute("Cost")
	end
	if v == nil then
		local def = HUB.getItemDef(item.Name)
		v = def and def.Attributes and def.Attributes.Value
	end
	return tonumber(v) or 0
end

-- 名字 → 中文（找不到时返回原名）
function HUB.zhName(name)
	if not name or name == "" then
		return "未知"
	end
	if exact[name] then
		return exact[name]
	end
	-- 去掉 Ore_ 前缀
	local stripped = string.match(name, "^Ore_(.+)$")
	if stripped and exact[stripped] then
		return exact[stripped]
	end
	-- 逐词翻译
	local parts = splitWords(name)
	local translated = {}
	local allHit = true
	for _, part in ipairs(parts) do
		if words[part] then
			table.insert(translated, words[part])
		elseif exact[part] then
			table.insert(translated, exact[part])
		else
			allHit = false
			break
		end
	end
	if allHit and #translated > 0 then
		return table.concat(translated, "")
	end
	return name
end

function HUB.zhOre(oreType)
	if not oreType then
		return "矿石"
	end
	return exact[tostring(oreType)] or HUB.zhName(tostring(oreType))
end

local function splitTokens(name)
	local out = {}
	for part in string.gmatch(tostring(name), "[^_]+") do
		local camel = string.gsub(part, "(%l)(%u)", "%1 %2")
		for word in string.gmatch(camel, "%S+") do
			word = string.gsub(word, "%d+$", "")
			if word ~= "" then
				table.insert(out, word)
			end
		end
	end
	return out
end

-- 让某个 Frame 可以拖动（触摸 + 鼠标）
local function makeDraggable(frame, handle, bounds)
	local dragging = false
	local dragStart, startPos
	handle = handle or frame

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not dragging then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)
end

function Window:tab(name, order)
	local btn = new("TextButton", {
		Size = UDim2.new(1, 0, 0, 32),
		BackgroundColor3 = T.Card,
		BackgroundTransparency = 0.55,
		Text = name,
		Font = T.Font,
		TextColor3 = T.TextDim,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left,
		AutoButtonColor = false,
		LayoutOrder = order or (#self.Tabs + 1),
		Parent = self.Rail,
	})
	corner(btn, 7)
	local bar = new("Frame", {
		Size = UDim2.new(0, 3, 0, 14),
		Position = UDim2.new(0, 4, 0.5, -7),
		BackgroundColor3 = T.Accent,
		BorderSizePixel = 0,
		BackgroundTransparency = 1,
		Parent = btn,
	})
	new("UIPadding", { PaddingLeft = UDim.new(0, 12), Parent = btn })

	local page = new("ScrollingFrame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 3,
		ScrollBarImageColor3 = T.Stroke,
		CanvasSize = UDim2.new(0, 0, 0, 0),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		ScrollingDirection = Enum.ScrollingDirection.Y,
		ElasticBehavior = Enum.ElasticBehavior.WhenScrollable,
		Visible = false,
		Parent = self.Content,
	})
	new("UIListLayout", {
		SortOrder = Enum.SortOrder.LayoutOrder,
		Padding = UDim.new(0, 10),
		Parent = page,
	})
	new("UIPadding", { PaddingBottom = UDim.new(0, 18), PaddingRight = UDim.new(0, 8), Parent = page })

	local tab = {
		Name = name,
		Page = page,
		Button = btn,
		Window = self,
		Sections = {},
	}

	function tab:section(title, order)
		local holder = new("Frame", {
			Size = UDim2.new(1, 0, 0, 0),
			BackgroundTransparency = 1,
			AutomaticSize = Enum.AutomaticSize.Y,
			LayoutOrder = order or (#self.Sections + 1),
			Parent = page,
		})
		new("UIListLayout", {
			SortOrder = Enum.SortOrder.LayoutOrder,
			Padding = UDim.new(0, 5),
			Parent = holder,
		})
		if title then
			local headerRow = new("Frame", {
				Size = UDim2.new(1, 0, 0, 16),
				BackgroundTransparency = 1,
				Parent = holder,
			})
			new("TextLabel", {
				Size = UDim2.new(1, -8, 1, 0),
				Position = UDim2.new(0, 8, 0, 0),
				BackgroundTransparency = 1,
				Font = T.FontBold,
				Text = title,
				TextColor3 = T.TextDim,
				TextSize = 11,
				TextXAlignment = Enum.TextXAlignment.Left,
				Parent = headerRow,
			})
		end
		local section = setmetatable({
			Holder = holder,
			Tab = self,
			holder = holder,
		}, Section)
		table.insert(self.Sections, section)
		return section
	end

	function tab:select()
		for _, other in ipairs(self.Window.Tabs) do
			if other == self then
				other.Page.Visible = true
				other.Button.BackgroundTransparency = 0
				other.Button.TextColor3 = T.Text
				other.Button.BackgroundColor3 = T.Accent
				other.Button.BackgroundTransparency = 0.75
				if other._bar then
					other._bar.BackgroundTransparency = 0
				end
			else
				other.Page.Visible = false
				other.Button.BackgroundTransparency = 0.55
				other.Button.TextColor3 = T.TextDim
				other.Button.BackgroundColor3 = T.Card
				if other._bar then
					other._bar.BackgroundTransparency = 1
				end
			end
		end
		self.Window.ActiveTab = self
	end

	tab._bar = bar
	btn.MouseButton1Click:Connect(function()
		tab:select()
	end)
	btn.MouseEnter:Connect(function()
		if self.ActiveTab ~= tab then
			tw(btn, 0.12, { BackgroundTransparency = 0.3 })
		end
	end)
	btn.MouseLeave:Connect(function()
		if self.ActiveTab ~= tab then
			tw(btn, 0.12, { BackgroundTransparency = 0.55 })
		end
	end)

	table.insert(self.Tabs, tab)
	tab:select()
	return tab
end

function Section:toggle(opts)
	local card = self:addCard(44)
	local on = opts.Default == true
	HUB.Flags[opts.Flag or opts.Text] = on

	new("TextLabel", {
		Size = UDim2.new(1, -70, 1, 0),
		Position = UDim2.new(0, 12, 0, 0),
		BackgroundTransparency = 1,
		Font = T.Font,
		Text = opts.Text or "",
		TextColor3 = T.Text,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Parent = card,
	})

	local back = new("Frame", {
		Size = UDim2.new(0, 38, 0, 20),
		Position = UDim2.new(1, -50, 0.5, -10),
		BackgroundColor3 = on and T.Accent or T.Stroke,
		BorderSizePixel = 0,
		Parent = card,
	})
	corner(back, 10)
	local knob = new("Frame", {
		Size = UDim2.new(0, 14, 0, 14),
		Position = UDim2.new(0, on and 22 or 3, 0.5, -7),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BorderSizePixel = 0,
		Parent = back,
	})
	corner(knob, 7)

	local btn = new("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = "",
		Parent = card,
	})

	local function apply(value, fire)
		on = value
		HUB.Flags[opts.Flag or opts.Text] = value
		tw(back, 0.15, { BackgroundColor3 = value and T.Accent or T.Stroke })
		tw(knob, 0.15, { Position = UDim2.new(0, value and 22 or 3, 0.5, -7) })
		if fire and opts.Callback then
			task.spawn(opts.Callback, value)
		end
	end

	btn.MouseButton1Click:Connect(function()
		apply(not on, true)
	end)

	if opts.Callback then
		task.spawn(opts.Callback, on)
	end

	local api = {}
	function api:set(value)
		apply(value, true)
	end
	function api:get()
		return on
	end
	return api
end

function Section:slider(opts)
	local card = self:addCard(52)
	local minV = opts.Min or 0
	local maxV = opts.Max or 100
	local value = math.clamp(opts.Default or minV, minV, maxV)
	HUB.Flags[opts.Flag or opts.Text] = value

	new("TextLabel", {
		Size = UDim2.new(1, -80, 0, 16),
		Position = UDim2.new(0, 12, 0, 7),
		BackgroundTransparency = 1,
		Font = T.Font,
		Text = opts.Text or "",
		TextColor3 = T.Text,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left,
		Parent = card,
	})
	local valueLabel = new("TextLabel", {
		Size = UDim2.new(0, 60, 0, 16),
		Position = UDim2.new(1, -68, 0, 7),
		BackgroundTransparency = 1,
		Font = T.FontBold,
		Text = tostring(value),
		TextColor3 = T.Accent,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Right,
		Parent = card,
	})

	local track = new("Frame", {
		Size = UDim2.new(1, -24, 0, 5),
		Position = UDim2.new(0, 12, 0, 34),
		BackgroundColor3 = T.Stroke,
		BorderSizePixel = 0,
		Parent = card,
	})
	corner(track, 3)
	local fill = new("Frame", {
		Size = UDim2.new((value - minV) / (maxV - minV), 0, 1, 0),
		BackgroundColor3 = T.Accent,
		BorderSizePixel = 0,
		Parent = track,
	})
	corner(fill, 3)

	local hitArea = new("TextButton", {
		Size = UDim2.new(1, -24, 0, 26),
		Position = UDim2.new(0, 12, 0, 24),
		BackgroundTransparency = 1,
		Text = "",
		Parent = card,
	})

	local function setValue(v, fire)
		value = math.clamp(v, minV, maxV)
		if opts.Round then
			value = math.floor(value + 0.5)
		end
		HUB.Flags[opts.Flag or opts.Text] = value
		fill.Size = UDim2.new((value - minV) / (maxV - minV), 0, 1, 0)
		valueLabel.Text = tostring(value)
		if fire and opts.Callback then
			task.spawn(opts.Callback, value)
		end
	end

	local dragging = false
	local function updateFromInput(input)
		local abs = track.AbsolutePosition
		local width = track.AbsoluteSize.X
		local alpha = (input.Position.X - abs.X) / math.max(width, 1)
		setValue(minV + (maxV - minV) * math.clamp(alpha, 0, 1), true)
	end

	hitArea.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			updateFromInput(input)
		end
	end)
	HUB.connect(UserInputService.InputChanged, function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			updateFromInput(input)
		end
	end)
	HUB.connect(UserInputService.InputEnded, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	if opts.Callback then
		task.spawn(opts.Callback, value)
	end

	local api = {}
	function api:set(v)
		setValue(v, true)
	end
	function api:get()
		return value
	end
	return api
end

function Section:divider()
	local card = self:addCard(1)
	card.BackgroundColor3 = T.Stroke
	card.BackgroundTransparency = 0.4
	return card
end

	})
	new("UIPadding", {
		PaddingTop = UDim.new(0, 4),
		PaddingBottom = UDim.new(0, 4),
		Parent = list,
	})

	local expanded = false
	local rowHeight = 26

	local function buildRows(listValues)
		for _, child in ipairs(list:GetChildren()) do
			if child:IsA("TextButton") then
				child:Destroy()
			end
		end
		for i, v in ipairs(listValues) do
			local row = new("TextButton", {
				Size = UDim2.new(1, 0, 0, rowHeight),
				BackgroundTransparency = 1,
				Text = "   " .. tostring(v),
				Font = T.Font,
				TextColor3 = T.TextDim,
				TextSize = 12,
				TextXAlignment = Enum.TextXAlignment.Left,
				LayoutOrder = i,
				AutoButtonColor = false,
				Parent = list,
			})
			row.MouseEnter:Connect(function()
				tw(row, 0.1, { BackgroundTransparency = 0.85, BackgroundColor3 = T.Accent })
			end)
			row.MouseLeave:Connect(function()
				tw(row, 0.1, { BackgroundTransparency = 1 })
			end)

				if opts.Callback then
					task.spawn(opts.Callback, v)
				end
			end)
		end
	end
	buildRows(values)

	local btn = new("TextButton", {
		Size = UDim2.new(1, 0, 0, 40),
		BackgroundTransparency = 1,
		Text = "",
		Parent = card,
	})
	btn.MouseButton1Click:Connect(function()
		expanded = not expanded
		local target = expanded and (#(HUB.Flags[opts.Text .. "_values"] or values) * (rowHeight + 2) + 8) or 0
		card.Size = UDim2.new(1, 0, 0, 40 + target)
		list.Size = UDim2.new(1, -16, 0, target)
	end)

	HUB.Flags[opts.Flag or opts.Text] = current
	if opts.Callback then
		task.spawn(opts.Callback, current)
	end

	local api = {}
	function api:set(v)

			task.spawn(opts.Callback, v)
		end
	end
	function api:refresh(listValues)
		local keep = current
		values = listValues
		buildRows(listValues)
		for _, v in ipairs(listValues) do
			if v == keep then
				current = keep
				valueLabel.Text = tostring(keep)
				return
			end
		end
		current = listValues[1]
		valueLabel.Text = tostring(current)
	end
	function api:refreshValues(listValues)
		values = listValues
		buildRows(listValues)
	end
	function api:get()
		return current
	end
	return api
end

-- ===== 15_toolhook.luau =====
do
--[[
	钻探至地球核心 · 客户端工具对象捕获
	游戏的自定义挥砍只认真实输入；拿到客户端工具对象后可以程序化挥砍
	（实测：BeginMeleeSwing + ResolveMeleeSwing 能真挖掉矿）
]]

local lp = game:GetService("Players").LocalPlayer

HUB.ClientTools = HUB.ClientTools or {}
getgenv().__earthTools = getgenv().__earthTools or {}

-- 上次会话捕获到的对象，直接接回来（重载不丢）
for kind, inst in pairs(getgenv().__earthTools) do
	if type(inst) == "table" and inst.Tool then
		HUB.ClientTools[kind] = inst
	end
end
if type(getgenv().__cap) == "table" and getgenv().__cap.Tool then
	HUB.ClientTools["Pickaxe"] = HUB.ClientTools["Pickaxe"] or getgenv().__cap
end

local function remember(kind, inst)
	if not inst then
		return
	end
	HUB.ClientTools[kind] = inst
	if inst.Tool then
		HUB.ClientTools[inst.Tool.Name] = inst
		getgenv().__earthTools[inst.Tool.Name] = inst
	end
	getgenv().__earthTools[kind] = inst
end

local MODULES = {
	Pickaxe = "Pickaxe",
	Sword = "Sword",
	Champions_Hammer = "Champions_Hammer",
}

local function installHook(kind, moduleName)
	local ok, mod = pcall(require, lp.PlayerScripts.SharedPlanets.Classes.Tools[moduleName])
	if not ok or type(mod) ~= "table" then
		return false
	end
	if mod.__earthHook then
		return true
	end
	local origHit = mod.HitAtSlashStart
	if type(origHit) == "function" then
		mod.HitAtSlashStart = function(self, id)
			remember(kind, self)
			return origHit(self, id)
		end
	end
	local origNew = mod.new
	if type(origNew) == "function" then
		mod.new = function(...)
			local inst = origNew(...)
			remember(kind, inst)
			return inst
		end
	end
	mod.__earthHook = true
	return true
end

for kind, moduleName in pairs(MODULES) do
	pcall(installHook, kind, moduleName)
end

-- 拿某个工具类别的客户端对象（没有就返回 nil）
function HUB.getToolInstance(kind)
	if kind == "Pickaxe" then
		for name, inst in pairs(HUB.ClientTools) do
			if string.find(name, "Pickaxe") and inst.Tool and inst.Tool.Parent == lp.Character then
				return inst
			end
		end
		for _, key in ipairs({ "Pickaxe", "Pickaxe_Wooden", "Pickaxe_Rusty", "Pickaxe_Steel", "Pickaxe_Gold", "Pickaxe_Diamond" }) do
			local inst = HUB.ClientTools[key]
			if inst and inst.Tool and inst.Tool.Parent == lp.Character then
				return inst
			end
		end
	elseif kind == "Sword" then
		for name, inst in pairs(HUB.ClientTools) do
			if string.find(name, "Sword") and inst.Tool and inst.Tool.Parent == lp.Character then
				return inst
			end
		end
	end
	local inst = HUB.ClientTools[kind]
	if inst and inst.Tool and inst.Tool.Parent == lp.Character then
		return inst
	end
	return nil
end

-- 程序化挥一下（先 BeginSwing 握手，再 Resolve 触发命中）
function HUB.clientSwing(kind, waitBeforeResolve)
	local inst = HUB.getToolInstance(kind)
	if not inst then
		return false
	end
	local ok = pcall(function()
		inst:BeginMeleeSwing()
	end)
	if not ok then
		return false
	end
	task.wait(waitBeforeResolve or 0.12)
	pcall(function()
		inst:ResolveMeleeSwing()
	end)
	return true
end

end

local flyConn = nil
local flyBV = nil
S.FlySpeed = 60

function HUB.setFly(enabled)
	S.Fly = enabled
	local char = HUB.Player.Character
	if not char then
		return
	end
	local root = HUB.rootPart()
	local hum = HUB.humanoid()
	if not root or not hum then
		return
	end
	if enabled then
		if flyBV then
			flyBV:Destroy()
		end
		flyBV = Instance.new("BodyVelocity")
		flyBV.Name = "EarthHubFly"
		flyBV.MaxForce = Vector3.new(1, 1, 1) * 1e5
		flyBV.Velocity = Vector3.zero
		flyBV.Parent = root
		if flyConn then
			flyConn:Disconnect()
		end
		flyConn = RunService.RenderStepped:Connect(function()
			local r = HUB.rootPart()
			if not r or not S.Fly then
				return
			end
			local cam = Workspace.CurrentCamera
			if not cam then
				return
			end
			local move = Vector3.zero
			local forward = cam.CFrame.LookVector
			local right = cam.CFrame.RightVector
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then
				move = move + forward
			end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then
				move = move - forward
			end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then
				move = move + right
			end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then
				move = move - right
			end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
				move = move + Vector3.new(0, 1, 0)
			end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
				move = move - Vector3.new(0, 1, 0)
			end
			-- 移动端：摇杆方向也参与飞行
			if move.Magnitude < 0.01 and UserInputService.TouchEnabled then
				if hum.MoveDirection.Magnitude > 0.05 then
					move = hum.MoveDirection
				end
			end
			if flyBV then
				flyBV.Velocity = move.Unit * S.FlySpeed * (move.Magnitude > 0 and 1 or 0)
			end
		end)
		pcall(function()
			hum.PlatformStand = true
		end)
	else
		if flyConn then
			flyConn:Disconnect()
			flyConn = nil
		end
		if flyBV then
			flyBV:Destroy()
			flyBV = nil
		end
		pcall(function()
			hum.PlatformStand = false
		end)
	end
end

-- 防摔死 / 血量保护（客户端，尽量不让传送把自己摔死）

S.FallProtect = false
S.HealthLock = false

HUB.connect(RunService.Heartbeat, function()
	local char = HUB.Player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not hum or not root then
		return
	end
	if S.FallProtect then
		local v = root.Velocity
		if v.Y < -90 then
			-- 快落地时刹车，避免摔死
			local params = RaycastParams.new()
			params.FilterDescendantsInstances = { char }
			params.FilterType = Enum.RaycastFilterType.Exclude
			local hit = Workspace:Raycast(root.Position, Vector3.new(0, -60, 0), params)
			if hit and (root.Position.Y - hit.Position.Y) < 30 then
				root.Velocity = Vector3.new(v.X * 0.2, -8, v.Z * 0.2)
			end
		end
	end
	if S.HealthLock and hum.Health < hum.MaxHealth then
		pcall(function()
			hum.Health = hum.MaxHealth
		end)
	end
end)

local function ensureEntry(model, part, color, name, distance)
	local entry = highlights[model]
	if not entry then
		entry = {}
		highlights[model] = entry
	end

	if S.ESP.Highlight then
		if not entry.hl or not entry.hl.Parent then
			pcall(function()
				entry.hl = makeHighlight(model, color)
			end)
		end
		if entry.hl then
			entry.hl.FillColor = color
			entry.hl.OutlineColor = color
			entry.hl.FillTransparency = S.ESP.Filled and S.ESP.FillTransparency or 1
		end
	elseif entry.hl then
		pcall(function()
			entry.hl:Destroy()
		end)
		entry.hl = nil
	end

	local hasName = (name ~= nil and name ~= "")
	local hasDist = (distance ~= nil and distance ~= "") and S.ESP.ShowDistance
	if hasName or hasDist then
		if not entry.billboard and part then
			pcall(function()
				entry.billboard, entry.nameLabel, entry.distLabel = makeLabel(part)
			end)
		end
		if entry.billboard then
			entry.billboard.Enabled = true
			entry.nameLabel.Visible = hasName
			entry.nameLabel.Text = name or ""
			entry.nameLabel.TextColor3 = color
			entry.nameLabel.Size = UDim2.new(1, 0, 0, hasName and 18 or 0)
			entry.distLabel.Visible = hasDist
			entry.distLabel.Text = distance or ""
			entry.distLabel.Position = UDim2.new(0, 0, 0, hasName and 18 or 0)
			entry.distLabel.Size = UDim2.new(1, 0, 0, hasDist and 14 or 0)
		end
	elseif entry.billboard then
		entry.billboard.Enabled = false
	end

	if not S.ESP.Highlight and (not entry.billboard or not entry.billboard.Enabled) then
		destroyEntry(model)
	end
end

---------------------------------------------------------------------
local function collect()
	local list = {}
	S.Values.DrawTotal = 0
	if not S.ESP.DrawEnabled then
		return list
	end
	local myRoot = HUB.rootPart()
	if not myRoot then
		return list
	end

	if S.ESP.Boss then
		for _, b in ipairs(HUB.getBosses(2000)) do
			if b.dist <= S.ESP.MaxDistance * 2 then
				list[#list + 1] = {
					model = b.model, part = b.part, color = COLORS.Boss,
					text = HUB.zhName(b.model.Name), dist = b.dist, nameable = true,
				}
			end
		end
	end

	if S.ESP.Enemies then
		for _, e in ipairs(HUB.getEnemies(S.ESP.MaxDistance)) do
			if e.hum.MaxHealth < 2000000 then
				list[#list + 1] = {
					model = e.model, part = e.part, color = COLORS.Enemy,
					text = HUB.zhName(e.model.Name), dist = e.dist, nameable = true,
				}
			end
		end
	end

	if S.ESP.Ores then
		for _, o in ipairs(HUB.getOres(S.ESP.OreDistance)) do
			local zhType = HUB.zhOre(o.oreType)
			if inSet(S.ESP.OreFilter, zhType) then
				list[#list + 1] = {
					model = o.part, part = o.part, color = COLORS.Ore,
					text = zhType, dist = o.dist, nameable = true,
				}
			end
		end
	end

	do
		for _, it in ipairs(HUB.getDroppedItems(S.ESP.MaxDistance)) do
			local zh = HUB.zhName(it.name)
			local cat = HUB.itemCategory(it.name)
			local catFilter = S.ESP.Filters[cat]
			if S.ESP.Cats[cat] and (catFilter == nil or next(catFilter) == nil or catFilter[zh] == true) then
				local color = COLORS.Item
				if cat == "武器" then
					color = COLORS.ItemWeapon
				elseif cat == "镐子" then
					color = COLORS.ItemPickaxe
				elseif cat == "医疗包" then
					color = COLORS.ItemMed
				elseif cat == "药水" then
					color = COLORS.ItemPotion
				elseif cat == "子弹" then
					color = COLORS.ItemAmmo
				elseif cat == "装备" then
					color = COLORS.ItemGear
				elseif cat == "卖钱" then
					color = COLORS.ItemValuable
				end
				if cat == "卖钱" then
					local value = HUB.itemValue(it.model)
					S.Values.DrawTotal = S.Values.DrawTotal + value
					if value > 0 then
						zh = zh .. " " .. HUB.formatNum(value)
					end
				end
				list[#list + 1] = { model = it.model, part = it.part, color = color, text = zh, dist = it.dist, nameable = true }
			end
		end
	end

	if S.ESP.Chests then
		for _, c in ipairs(HUB.getChests(S.ESP.MaxDistance)) do
			local tier = HUB.chestTier(c.name)
			local zhChest = HUB.zhName(c.name)
			if inSet(S.ESP.ChestFilter, zhChest) then
				local color = (tier == "高级") and COLORS.ChestHigh or COLORS.ChestLow
				list[#list + 1] = {
					model = c.model, part = c.part, color = color,
					text = zhChest, dist = c.dist, nameable = true,
				}
			end
		end
	end

	if S.ESP.Gates then
		for _, g in ipairs(HUB.getGates()) do
			if g.dist <= S.ESP.MaxDistance * 2 then
				list[#list + 1] = {
					model = g.model, part = g.part, color = COLORS.Gate,
					text = "第" .. tostring(g.tier) .. "层地牢", dist = g.dist, nameable = true,
				}
			end
		end
	end

	return list
end

local function refresh()
	local list = collect()
	local seen = {}
	for _, item in ipairs(list) do
		local nameText = item.text
		local distText = math.floor(item.dist) .. " 米"
		ensureEntry(item.model, item.part, item.color, nameText, distText)
		seen[item.model] = true
	end
	for model in pairs(highlights) do
		if not seen[model] or not model.Parent then
			destroyEntry(model)
		end
	end
end

	local now = os.clock()
	if now - refreshAt < 0.3 then
		return
	end
	refreshAt = now
	local ok = pcall(refresh)
	if not ok then
		for model in pairs(highlights) do
			destroyEntry(model)
		end
	end
end)

local function finiteVec(v)
	if not v then
		return false
	end
	local x, y, z = v.X, v.Y, v.Z
	if x ~= x or y ~= y or z ~= z then
		return false
	end
	if math.abs(x) > 1e6 or math.abs(y) > 1e6 or math.abs(z) > 1e6 then
		return false
	end
	return true
end

function HUB.teleportTo(cf)
	local root = HUB.rootPart()
	if not root or not cf then
		return false
	end
	if not finiteVec(cf.Position) then
		return false
	end
	local hum = HUB.humanoid()
	if hum then
		pcall(function()
			root.CFrame = cf
		end)
		pcall(function()
			root.Velocity = Vector3.zero
		end)
		return true
	end
	return false
end

function HUB.teleportToPosition(pos)
	local root = HUB.rootPart()
	if not root or not finiteVec(pos) then
		return false
	end
	return HUB.teleportTo(CFrame.new(pos + Vector3.new(0, 3, 0)))
end

function HUB.teleportToPart(part)
	if not part or not finiteVec(part.Position) then
		return false
	end
	local root = HUB.rootPart()
	if not root then
		return false
	end
	local offset = (root.Position - part.Position).Unit * 4
	if offset.Magnitude < 0.1 then
		offset = Vector3.new(0, 0, 4)
	end
	return HUB.teleportTo(CFrame.new(part.Position + offset + Vector3.new(0, 2, 0)))
end

function HUB.teleportToPlayer(name)
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Name:lower() == tostring(name):lower() and plr.Character then
			local root = plr.Character:FindFirstChild("HumanoidRootPart")
			if root then
				return HUB.teleportTo(root.CFrame * CFrame.new(0, 0, 3))
			end
		end
	end
	return false
end

-- 相机射线取点
function HUB.cameraRaycast(maxDistance)
	local cam = Workspace.CurrentCamera
	if not cam then
		return nil
	end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	local char = HUB.Player.Character
	params.FilterDescendantsInstances = char and { char } or {}
	local result = Workspace:Raycast(cam.CFrame.Position, cam.CFrame.LookVector * (maxDistance or 2000), params)
	return result and result.Position or nil
end

-- 点击传送
HUB.connect(UserInputService.InputBegan, function(input, processed)
	if not S.ClickTP or processed then
		return
	end
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		local pos = HUB.cameraRaycast(3000)
		if pos then
			HUB.teleportToPosition(pos)
			HUB.Notify("已传送到准星位置", "ok", 1.5)
		end
	end
end)

		return false
	end

end

		return false
	end

local S = HUB.State
local LP = HUB.Player

S.KillAura = {
	Enabled = false,
	Range = 18,      -- 近战判定大概 18 格
	Delay = 0.15,
	Teleport = false,
	OnlyBoss = false,

local weaponRank = {
	Sword_Phoenix = 100,
	Sword_Void = 90,
	Sword_Diamond = 85,
	Sword_Gold = 80,
	Sword_Steel = 70,
	Sword_Rusty = 60,

local function rankOf(tool)
	if not tool then
		return -1
	end
	return weaponRank[tool.Name] or 0
end

-- 背包里最好的武器
function HUB.bestWeapon()
	local bp = LP:FindFirstChildOfClass("Backpack")
	if not bp then
		return nil
	end
	local best, bestRank = nil, -1
	for _, item in ipairs(bp:GetChildren()) do
		if item:IsA("Tool") then
			local r = rankOf(item)
			if string.find(item.Name, "Sword") then
				r = r + 5
			end
			if r > bestRank then
				best, bestRank = item, r
			end
		end
	end
	return best
end

-- 手上没武器就从背包拿一把（有武器就不动，避免抖）
function HUB.equipWeapon()
	local char = LP.Character
	if not char then
		return nil
	end
	local equipped = char:FindFirstChildOfClass("Tool")
	if equipped and rankOf(equipped) > 0 then
		return equipped
	end
	local tool = HUB.bestWeapon()
	if tool then
		local hum = char:FindFirstChildOfClass("Humanoid")
		pcall(function()
			hum:EquipTool(tool)
		end)
		task.wait(0.1)
		return tool
	end
	return nil
end

-- 只挥，不动
local function auraLoop()
	local lastTeleport = 0
	while true do
		local ok = pcall(function()
			if not S.KillAura.Enabled or not HUB.isAlive() then
				task.wait(0.25)
				return
			end

			local target
			if S.KillAura.OnlyBoss then
				for _, b in ipairs(HUB.getBosses(2000)) do
					if b.dist <= S.KillAura.Range then
						target = b
						break
					end
				end
			else
				for _, e in ipairs(HUB.getEnemies(S.KillAura.Range)) do
					if e.hum.MaxHealth < 1000000 then
						target = e
						break
					end
				end
			end

			-- 范围内没怪就不挥，省得空挥乱抖
			if not target then
				task.wait(0.2)
				return
			end

			local tool = HUB.equipWeapon()
			if not tool then
				task.wait(0.4)
				return
			end

			-- 默认完全不碰位置和朝向；只有手动开了瞬移才挪一次
			if S.KillAura.Teleport then
				local now = os.clock()
				local root = HUB.rootPart()
				if root and now - lastTeleport > 1.5 then
					local dist = (target.part.Position - root.Position).Magnitude
					if dist > 10 then
						lastTeleport = now
						local pos = target.part.Position
						pcall(function()
							root.CFrame = CFrame.lookAt(pos + Vector3.new(0, 0, 6), pos)
						end)
						task.wait(0.2)
					end
				end
			end

			local swung = HUB.clientSwing("Sword", 0.1)
			if not swung then
				pcall(function()
					tool:Activate()
				end)
			end
			task.wait(S.KillAura.Delay)
		end)
		if not ok then
			task.wait(0.3)
		end
	end
end

task.spawn(function()
	pcall(auraLoop)
end)

end

---------------------------------------------------------------------
-- 发现 Knit 服务目录
---------------------------------------------------------------------
local cachedServices = nil
function HUB.getServices()
	if cachedServices and cachedServices.Parent then
		return cachedServices
	end
	cachedServices = nil
	local ok, result = pcall(function()
		for _, inst in ipairs(ReplicatedStorage:GetDescendants()) do
			if inst.Name == "Services" and inst.ClassName == "Folder" then
				local parent = inst.Parent
				if parent and (parent.Name == "knit" or (parent.Parent and parent.Parent.Name == "knit")) then
					return inst
				end
			end
		end
		return nil
	end)
	cachedServices = ok and result or nil
	return cachedServices
end

-- 拿某个服务下的远程对象
function HUB.getRemote(serviceName, remoteName)
	local services = HUB.getServices()
	if not services then
		return nil
	end
	local svc = services:FindFirstChild(serviceName)
	if not svc then
		return nil
	end
	for _, container in ipairs(svc:GetChildren()) do
		if container.Name == "RF" or container.Name == "RE" then
			local remote = container:FindFirstChild(remoteName)
			if remote then
				return remote
			end
		end
	end
	return nil
end

-- 安全调用 RemoteFunction：返回 ok, result
function HUB.invoke(serviceName, remoteName, ...)
	local remote = HUB.getRemote(serviceName, remoteName)
	if not remote or not remote:IsA("RemoteFunction") then
		return false, "远程不存在: " .. tostring(serviceName) .. "." .. tostring(remoteName)
	end
	local args = table.pack(...)
	local ok, result = pcall(function()
		return remote:InvokeServer(table.unpack(args, 1, args.n))
	end)
	return ok, result
end

function HUB.fire(serviceName, remoteName, ...)
	local remote = HUB.getRemote(serviceName, remoteName)
	if not remote or not remote:IsA("RemoteEvent") then
		return false, "事件不存在"
	end
	local args = table.pack(...)
	local ok, err = pcall(function()
		remote:FireServer(table.unpack(args, 1, args.n))
	end)
	return ok, err
end

-- 列出所有远程（用于面板）
function HUB.listRemotes()
	local services = HUB.getServices()
	local list = {}
	if not services then
		return list
	end
	for _, svc in ipairs(services:GetChildren()) do
		for _, container in ipairs(svc:GetChildren()) do
			if container.Name == "RF" or container.Name == "RE" then
				for _, remote in ipairs(container:GetChildren()) do
					table.insert(list, svc.Name .. "." .. remote.Name)
				end
			end
		end
	end
	table.sort(list)
	return list
end

---------------------------------------------------------------------
-- 自动交互（宝箱 / 门 / 机器 等 ProximityPrompt）
---------------------------------------------------------------------
local lastInteract = 0
HUB.connect(game:GetService("RunService").Heartbeat, function()
	if not S.AutoInteract.Enabled then
		return
	end
	local now = os.clock()
	if now - lastInteract < S.AutoInteract.Delay then
		return
	end
	local root = HUB.rootPart()
	if not root then
		return
	end
	local ok = pcall(function()
		for _, prompt in ipairs(Workspace:GetDescendants()) do
			if prompt:IsA("ProximityPrompt") and prompt.Enabled then
				local parent = prompt.Parent
				if parent and parent:IsA("BasePart") then
					local dist = (parent.Position - root.Position).Magnitude
					if dist <= math.min(S.AutoInteract.Distance, prompt.MaxActivationDistance + 2) then
						if type(fireproximityprompt) == "function" then
							fireproximityprompt(prompt)
						else
							prompt:InputHoldBegin()
							task.wait(0.05)
							prompt:InputHoldEnd()
						end
						lastInteract = now
						return
					end
				end
			end
		end
	end)
	if not ok then
		-- 出错时不刷屏
		lastInteract = now
	end
end)

-- ===== 55_mining.luau =====
do
--[[
	钻探至地球核心 · 自动挖矿 + 全图拾取
	挖矿：传送到矿点 → 换镐 → 面向 → Activate 若干次 → 下一块
	拾取：传送到掉落物 → CarryInventoryService.Pickup
]]

local S = HUB.State
local LP = HUB.Player

S.AutoMine = {
	Enabled = false,
	OreType = "全部",
	Delay = 0.15,        -- 固定最小敲击间隔
	NoclipOwned = false, -- 穿墙是本功能开的还是用户自己开的
	MaxDistance = 3000,  -- 固定最大找矿半径
	WaitTimeout = 25,    -- 等一块矿挖完的最长时间
}

S.AutoPickup = {
	Enabled = false,
	MaxDistance = 4000,
	Delay = 0.35,
	Cats = {          -- 捡哪些分类
		["武器"] = true,
		["镐子"] = true,
		["医疗包"] = true,
		["药水"] = true,
		["子弹"] = true,
		["装备"] = true,
		["卖钱"] = true,
		["尸体"] = true,
		["其他"] = true,
	},
	Filters = {},     -- [分类] = { [中文名] = true }；空 = 该类全捡
}

local blacklist = {}   -- [part] = 解禁时间
local stats = { mined = 0, picked = 0 }

function HUB.getMiningStats()
	return stats.mined, stats.picked
end

-- 换镐：一定要确认真的拿在手上，拿不上就重试
local function pickaxe()
	local char = LP.Character
	if not char then
		return nil
	end
	local function equippedTool()
		return char:FindFirstChildOfClass("Tool")
	end
	local function isPickaxe(t)
		return t and string.find(t.Name, "Pickaxe") ~= nil
	end

	if isPickaxe(equippedTool()) then
		return equippedTool()
	end

	local bp = LP:FindFirstChildOfClass("Backpack")
	if not bp then
		return nil
	end
	local best, bestRank = nil, -1
	local ranks = {
		Pickaxe_Phoenix = 90, Pickaxe_Void = 80, Pickaxe_Diamond = 70,
		Pickaxe_Gold = 60, Pickaxe_Steel = 50, Pickaxe_Rusty = 40, Pickaxe_Wooden = 30,
		Pickaxe_Prospector = 45,
	}
	for _, item in ipairs(bp:GetChildren()) do
		if item:IsA("Tool") and string.find(item.Name, "Pickaxe") then
			local r = ranks[item.Name] or 10
			if r > bestRank then
				best, bestRank = item, r
			end
		end
	end
	if not best then
		return nil
	end

	local hum = char:FindFirstChildOfClass("Humanoid")
	for attempt = 1, 4 do
		pcall(function()
			if equippedTool() and not isPickaxe(equippedTool()) then
				hum:UnequipTools()
				task.wait(0.15)
			end
			hum:EquipTool(best)
		end)
		for _ = 1, 10 do
			task.wait(0.08)
			if isPickaxe(equippedTool()) then
				return equippedTool()
			end
		end
	end
	return isPickaxe(equippedTool()) and equippedTool() or nil
end

-- 拾取进背包；全程不替换手上的工具（收完再装回去）
-- 按当前筛选，找第一个该捡的物品，同时统计卖钱总价值
function HUB.findPickTarget()
	local items = HUB.getDroppedItems(S.AutoPickup.MaxDistance)
	local first, total = nil, 0
	for _, cand in ipairs(items) do
		local cat = HUB.itemCategory(cand.name)
		local zh = HUB.zhName(cand.name)
		local cf = S.AutoPickup.Filters[cat]
		local catOk = S.AutoPickup.Cats[cat] ~= false
		local nameOk = (cf == nil or next(cf) == nil) or cf[zh] == true
		if catOk and nameOk then
			if cat == "卖钱" then
				total = total + HUB.itemValue(cand.model)
			end
			if not first then
				first = cand
			end
		end
	end
	S.Values = S.Values or {}
	S.Values.PickTotal = total
	return first
end

function HUB.pickupItem(item)
	local char = LP.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local held = char and char:FindFirstChildOfClass("Tool")
	local ok, res = HUB.invoke("CarryInventoryService", "Pickup", item)
	if hum and held and held.Parent and char:FindFirstChildOfClass("Tool") ~= held then
		pcall(function()
			hum:EquipTool(held)
		end)
	end
	return ok and res == true or (ok and res)
end

---------------------------------------------------------------------
-- 主循环：拾取优先，其次挖矿
---------------------------------------------------------------------
task.spawn(function()
	while true do
		local ok, err = pcall(function()
			-- 挖矿关掉后，把本功能开的穿墙也收回去
			if not S.AutoMine.Enabled and S.AutoMine.NoclipOwned then
				S.AutoMine.NoclipOwned = false
				if S.Noclip then
					HUB.setNoclip(false)
				end
			end

			if not HUB.isAlive() then
				task.wait(0.4)
				return
			end

			-- 1) 全图拾取
			if S.AutoPickup.Enabled then
				local it = HUB.findPickTarget()
				if it then
					local myRoot = HUB.rootPart()
					pcall(function()
						myRoot.CFrame = CFrame.new(it.part.Position + Vector3.new(0, 1.2, 0))
					end)
					task.wait(0.08)
					if HUB.pickupItem(it.model) then
						stats.picked = stats.picked + 1
					end
					task.wait(S.AutoPickup.Delay)
					return
				end
			end

			-- 2) 自动挖矿
			if S.AutoMine.Enabled then
				local now = os.clock()
				local ores = HUB.getOres(S.AutoMine.MaxDistance, S.AutoMine.OreType)
				local target
				for _, ore in ipairs(ores) do
					local until_ = blacklist[ore.part]
					if not until_ or now > until_ then
						target = ore
						break
					end
				end
				if target then
					-- 先确保镐在手上（有时装备会失败）
					local ensured = pickaxe()
					if not ensured then
						task.wait(0.5)
						return
					end
					-- 挖矿期间开穿墙，钻到矿内部
					if not S.Noclip then
						HUB.setNoclip(true)
						S.AutoMine.NoclipOwned = true
					end
					local pick = pickaxe()
					if pick then
						local myRoot = HUB.rootPart()
						local orePos = target.part.Position
						pcall(function()
							myRoot.CFrame = CFrame.new(orePos + Vector3.new(0, 0.5, 0))
						end)
						-- 一块矿：一直敲到它消失（或超时）才换下一块
						local deadline = os.clock() + S.AutoMine.WaitTimeout
						while target.part.Parent and os.clock() < deadline do
							pcall(function()
								myRoot.CFrame = CFrame.new(orePos + Vector3.new(0, 0.5, 0))
							end)
							-- 优先用客户端工具对象真挥砍；没有就退回 Activate
							if not string.find((LP.Character:FindFirstChildOfClass("Tool") and LP.Character:FindFirstChildOfClass("Tool").Name) or "", "Pickaxe") then
								pickaxe()
							end
							local swung = HUB.clientSwing("Pickaxe", 0.1)
							if not swung then
								pcall(function()
									pick:Activate()
								end)
							end
							task.wait(S.AutoMine.Delay)
						end
						blacklist[target.part] = os.clock() + 20
						stats.mined = stats.mined + 1
					else
						task.wait(0.5)
					end
					return
				end
			end

			task.wait(0.35)
		end)
		if not ok then
			task.wait(0.5)
		end
	end
end)

end

-- ===== 60_nav.luau =====
do
--[[
	钻探至地球核心 · 自动寻路（去地牢门 / 进地牢 / 去 BOSS）
]]

local PathfindingService = game:GetService("PathfindingService")
local RunService        = game:GetService("RunService")

local S = HUB.State

S.Nav = {
	Active = false,
	Status = "空闲",
	Target = nil,
}

local navToken = 0

function HUB.stopNav()
	navToken = navToken + 1
	S.Nav.Active = false
	S.Nav.Status = "已停止"
	local hum = HUB.humanoid()
	if hum then
		pcall(function()
			hum:MoveTo(hum.Parent.HumanoidRootPart.Position)
		end)
	end
end

local function walkPath(targetPos, label)
	local myToken = navToken + 1
	navToken = myToken
	S.Nav.Active = true
	S.Nav.Status = "规划路径：" .. label
	S.Nav.Target = targetPos

	local path = PathfindingService:CreatePath({
		AgentRadius = 2,
		AgentHeight = 5,
		AgentCanJump = true,
		AgentCanClimb = true,
		AgentMaxSlope = 60,
	})

	local ok = pcall(function()
		path:ComputeAsync(HUB.rootPart().Position, targetPos)
	end)
	if not ok or path.Status ~= Enum.PathStatus.Success then
		S.Nav.Status = "寻路失败，尝试直走"
		local hum = HUB.humanoid()
		if hum then
			pcall(function()
				hum:MoveTo(targetPos)
			end)
		end
	else
		local waypoints = path:GetWaypoints()
		for i, wp in ipairs(waypoints) do
			if navToken ~= myToken then
				S.Nav.Active = false
				S.Nav.Status = "已停止"
				return false
			end
			local hum = HUB.humanoid()
			if not hum or hum.Health <= 0 then
				S.Nav.Active = false
				S.Nav.Status = "角色死亡"
				return false
			end
			S.Nav.Status = string.format("前往 %s（%d/%d）", label, i, #waypoints)
			hum:MoveTo(wp.Position)
			if wp.Action == Enum.PathWaypointAction.Jump then
				pcall(function()
					hum.Jump = true
				end)
			end
			-- 等待到达或超时
			local deadline = os.clock() + 4
			while os.clock() < deadline do
				if navToken ~= myToken then
					return false
				end
				local root = HUB.rootPart()
				if root and (root.Position - wp.Position).Magnitude < 4 then
					break
				end
				task.wait(0.1)
			end
		end
	end

	local root = HUB.rootPart()
	if root and (root.Position - targetPos).Magnitude < 12 then
		S.Nav.Active = false
		S.Nav.Status = "已到达：" .. label
		return true
	end
	S.Nav.Active = false
	S.Nav.Status = "接近：" .. label
	return true
end

function HUB.navigateToPos(pos, label)
	task.spawn(function()
		pcall(walkPath, pos, label or "目标点")
	end)
end

function HUB.navigateToNearestGate(tier)
	local gates = HUB.getGates()
	local target
	for _, g in ipairs(gates) do
		if not tier or g.tier == tier then
			target = g
			break
		end
	end
	if not target then
		HUB.Notify("没找到地牢门", "warn")
		return
	end
	HUB.navigateToPos(target.part.Position, "Tier" .. tostring(target.tier) .. " 地牢门")
	-- 到门口后自动按提示
	task.spawn(function()
		local deadline = os.clock() + 120
		while os.clock() < deadline and S.Nav.Active do
			task.wait(0.5)
		end
		local root = HUB.rootPart()
		if root and (root.Position - target.part.Position).Magnitude < 30 then
			local ok = HUB.firePrompt(target.part)
			HUB.Notify(ok and "已触发地牢入口" or "没找到入口提示，走近一点再试", ok and "ok" or "warn", 4)
		end
	end)
end

function HUB.enterNearestGate()
	local gates = HUB.getGates()
	local target = gates[1]
	if not target then
		HUB.Notify("附近没有地牢门", "warn")
		return
	end
	local root = HUB.rootPart()
	if root and (root.Position - target.part.Position).Magnitude > 30 then
		HUB.navigateToNearestGate()
		return
	end
	local ok = HUB.firePrompt(target.part)
	HUB.Notify(ok and "已触发地牢入口" or "没有可用的入口提示", ok and "ok" or "warn", 4)
end

-- 去 BOSS：有 BOSS 直接去，没有就去最近地牢
function HUB.navigateToBoss()
	local bosses = HUB.getBosses(2000)
	if #bosses > 0 then
		local b = bosses[1]
		HUB.navigateToPos(b.part.Position, "BOSS " .. b.model.Name)
		HUB.Notify("正在前往 BOSS：" .. b.model.Name, "ok", 3)
		return
	end
	HUB.Notify("当前区域没有 BOSS，改为前往最近的地牢门", "warn", 4)
	HUB.navigateToNearestGate()
end

end

-- ===== 65_vehicle.luau =====
do
--[[
	钻探至地球核心 · 钻机车 / 熔炉（烧尸体）
	车 = 带 VehicleSeat 的 Drill 模型；熔炉 = 车上的 Furnace 部件
]]

local Workspace = game:GetService("Workspace")

local S = HUB.State
local LP = HUB.Player

S.Vehicle = {
	BurnRunning = false,
}

-- 烧尸体可调参数
S.Burn = {
	Enabled = false,     -- 开关：开了就自动循环烧
	Max = 10,            -- 一次最多拿几个（背几具尸体）
	Radius = 4000,       -- 找尸体的半径
	Delay = 0.3,         -- 每烧一个的间隔
	UseVehicleFurnace = true, -- 先飞去钻机车上的熔炉
}

-- 兼容旧引用
S.Burn.Stats = 0

-- 所有钻机车（含各层堡垒里的）
function HUB.getDrills()
	local out = {}
	local myRoot = HUB.rootPart()
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("Model") and inst.Name == "Drill" then
			local seat = inst:FindFirstChildWhichIsA("VehicleSeat", true)
			local furnace = inst:FindFirstChild("Furnace", true)
			if seat then
				out[#out + 1] = {
					model = inst,
					seat = seat,
					furnace = furnace,
					dist = myRoot and (seat.Position - myRoot.Position).Magnitude or 0,
				}
			end
		end
	end
	-- 堡垒里的 Drill_Portal 也算
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("Model") and inst.Name == "Drill_Portal" then
			local seat = inst:FindFirstChildWhichIsA("VehicleSeat", true)
			if seat then
				out[#out + 1] = {
					model = inst,
					seat = seat,
					furnace = inst:FindFirstChild("Furnace", true),
					dist = myRoot and (seat.Position - myRoot.Position).Magnitude or 0,
				}
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

function HUB.teleportToVehicle()
	local drills = HUB.getDrills()
	local d = drills[1]
	if not d then
		HUB.Notify("没找到钻机车", "warn")
		return false
	end
	local myRoot = HUB.rootPart()
	if not myRoot then
		return false
	end
	pcall(function()
		myRoot.CFrame = d.seat.CFrame * CFrame.new(0, 3, 0)
		myRoot.Velocity = Vector3.zero
	end)
	HUB.Notify("已传送到钻机车（" .. math.floor(d.dist) .. " 米外）", "ok", 2)
	return true
end

-- 最近的熔炉部件
function HUB.getFurnace()
	local myRoot = HUB.rootPart()
	local best, bestDist = nil, math.huge
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("BasePart") and inst.Name == "Furnace" then
			if myRoot then
				local d = (inst.Position - myRoot.Position).Magnitude
				if d < bestDist then
					best, bestDist = inst, d
				end
			elseif not best then
				best = inst
			end
		end
	end
	if best then
		return best, bestDist
	end
	-- 退一步：车上的熔炉
	local drills = HUB.getDrills()
	for _, d in ipairs(drills) do
		if d.furnace then
			return d.furnace, d.dist
		end
	end
	return nil, 0
end

function HUB.teleportToFurnace()
	local furnace = HUB.getFurnace()
	if not furnace then
		HUB.Notify("没找到熔炉", "warn")
		return false
	end
	local myRoot = HUB.rootPart()
	if myRoot then
		pcall(function()
			myRoot.CFrame = CFrame.new(furnace.Position + Vector3.new(0, 4, 4))
		end)
		HUB.Notify("已传送到熔炉", "ok", 2)
		return true
	end
	return false
end

-- 尸体识别
---------------------------------------------------------------------
-- 死亡的单位模型（可背）
function HUB.getCorpses(maxDist)
	local out = {}
	local myRoot = HUB.rootPart()
	for _, inst in ipairs(Workspace:GetDescendants()) do
		if inst:IsA("Humanoid") and inst.Parent and inst.Health <= 0 then
			local model = inst.Parent
			if model ~= LP.Character then
				local part = model:FindFirstChildWhichIsA("BasePart", true)
				if part then
					local d = myRoot and (part.Position - myRoot.Position).Magnitude or 0
					if not maxDist or d <= maxDist then
						out[#out + 1] = { model = model, part = part, dist = d, name = model.Name }
					end
				end
			end
		end
	end
	table.sort(out, function(a, b)
		return a.dist < b.dist
	end)
	return out
end

-- 背包里的尸体（Items 文件夹里名字是怪名的）
function HUB.getLooseCorpses(maxDist)
	local out = {}
	for _, corpse in ipairs(HUB.getCorpses(maxDist)) do
		out[#out + 1] = corpse
	end
	return out
end

---------------------------------------------------------------------
-- 一键烧熔炉：背起尸体 → 飞到熔炉 → 塞进去
---------------------------------------------------------------------
function HUB.burnCorpses()
	if S.Vehicle.BurnRunning then
		HUB.Notify("烧炉子已经在跑了", "warn")
		return
	end
	S.Vehicle.BurnRunning = true
	task.spawn(function()
		local ok, err = pcall(function()
			local cfg = S.Burn
			local capacity = 35
			pcall(function()
				capacity = tonumber(LP:GetAttribute("CarryCapacity")) or 35
			end)
			local want = math.max(1, math.min(math.floor(cfg.Max), capacity))

			-- 1) 先背尸体：一次最多拿 want 个
			local carried, failed = 0, 0
			local myRoot = HUB.rootPart()
			for i = 1, want do
				local corpses = HUB.getCorpses(cfg.Radius)
				if #corpses == 0 then
					break
				end
				local c = corpses[1]
				myRoot = HUB.rootPart()
				if myRoot then
					pcall(function()
						myRoot.CFrame = CFrame.new(c.part.Position + Vector3.new(0, 1.5, 0))
					end)
					task.wait(0.12)
				end
				if HUB.invoke("CarryInventoryService", "Pickup", c.model) then
					carried = carried + 1
				else
					failed = failed + 1
					if failed >= 3 then
						break
					end
				end
				task.wait(cfg.Delay)
			end

			if carried == 0 then
				HUB.Notify("附近没有能背起来的尸体", "warn", 3)
				return
			end

			-- 2) 飞去熔炉，把背上的全塞进去
			if cfg.UseVehicleFurnace then
				HUB.teleportToVehicle()
			end
			local furnace = HUB.getFurnace()
			if not furnace then
				HUB.Notify("没找到熔炉", "err")
				return
			end
			myRoot = HUB.rootPart()
			if myRoot then
				pcall(function()
					myRoot.CFrame = CFrame.new(furnace.Position + Vector3.new(0, 4, 4))
				end)
			end
			task.wait(0.2)

			for i = 1, carried do
				HUB.invoke("InteractionService", "Interact", furnace)
				HUB.invoke("CarryInventoryService", "DropInventory")
				task.wait(cfg.Delay)
			end

			HUB.Notify("这趟背了 " .. carried .. " 具，已全部送进熔炉", "ok", 4)
		end)
		S.Vehicle.BurnRunning = false
		if not ok then
			HUB.Notify("烧炉出错：" .. tostring(err), "err", 5)
		end
	end)
end

end

local function httpGet(url)
	if type(game.HttpGet) ~= "function" then
		return nil
	end
	local ok, res = pcall(function()
		return game:HttpGet(url, true)
	end)
	if not ok then
		ok, res = pcall(function()
			return game:HttpGet(url)
		end)
	end
	if ok and type(res) == "string" and #res > 20000 then
		return res
	end
	return nil
end

function UIKit:init()
	for _, url in ipairs(OBSIDIAN_URLS) do
		local src = httpGet(url)
		if src then
			local okFn, fn = pcall(loadstring or load, src)
			if okFn and fn then
				local okRun, lib = pcall(fn)
				if okRun and lib and lib.CreateWindow then
					UIKit.Library = lib
					UIKit.Backend = "obsidian"
					return true
				end
			end
		end
	end
	UIKit.Backend = "native"
	return false
end

local uid = 0
local function key(prefix, opts)

	return prefix .. tostring(uid) .. "_" .. tostring(opts and opts.Text or "")

local function wrapGroup(group)
	local api = {}

	function api:toggle(opts)
		local state = {}
		local value = opts.Default == true
		HUB.Flags[opts.Flag or opts.Text] = value
		pcall(function()
			state._obj = group:AddToggle(key("tg", opts), {
				Text = opts.Text,
				Default = value,
				Callback = function(v)
					value = v
					HUB.Flags[opts.Flag or opts.Text] = v
					if opts.Callback then
						task.spawn(opts.Callback, v)
					end
				end,
			})
		end)
		function state:set(v)
			pcall(function()
				state._obj:SetValue(v)
			end)
		end
		function state:get()
			return value
		end
		if opts.Callback then
			task.spawn(opts.Callback, value)
		end
		return state
	end

	function api:slider(opts)
		local state = {}
		local value = opts.Default or opts.Min or 0
		HUB.Flags[opts.Flag or opts.Text] = value
		pcall(function()
			state._obj = group:AddSlider(key("sl", opts), {
				Text = opts.Text,
				Default = value,
				Min = opts.Min or 0,
				Max = opts.Max or 100,
				Rounding = 0,
				Callback = function(v)
					value = v
					HUB.Flags[opts.Flag or opts.Text] = v
					if opts.Callback then
						task.spawn(opts.Callback, v)
					end
				end,
			})
		end)
		function state:set(v)
			pcall(function()
				state._obj:SetValue(v)
			end)
		end
		function state:get()
			return value
		end
		if opts.Callback then
			task.spawn(opts.Callback, value)
		end
		return state
	end

	function api:button(opts)
		pcall(function()
			group:AddButton({
				Text = opts.Text,
				Func = function()
					if opts.Callback then
						local ok, err = pcall(opts.Callback)
						if not ok then
							HUB.Notify(tostring(err), "err", 4)
						end
					end
				end,
			})
		end)
	end

	function api:dropdown(opts)
		local state = {}
		local isMulti = opts.Multi == true
		local current
		if isMulti then
			current = {}
		else
			current = opts.Default or (opts.Values and opts.Values[1])
		end
		pcall(function()
			state._obj = group:AddDropdown(key("dd", opts), {
				Text = opts.Text,
				Values = opts.Values or {},
				Default = current,
				Multi = isMulti,
				Searchable = opts.Searchable == true,
				Callback = function(v)
					current = v
					HUB.Flags[opts.Flag or opts.Text] = v
					if opts.Callback then
						task.spawn(opts.Callback, v)
					end
				end,
			})
		end)
		function state:get()
			return current
		end
		function state:set(v)
			current = v
			pcall(function()
				state._obj:SetValue(v)
			end)
		end
		function state:refresh(values)
			pcall(function()
				state._obj:SetValues(values)
			end)
			pcall(function()
				state._obj:SetValue(values[1])
			end)
			current = values[1]
		end
		-- 换选项但保留已选（多选筛选用）
		function state:refreshValues(values)
			pcall(function()
				state._obj:SetValues(values)
			end)
		end
		return state
	end

	function api:input(opts)
		local state = {}
		local text = ""
		pcall(function()
			state._obj = group:AddInput(key("in", opts), {
				Text = opts.Text,
				Default = "",
				Numeric = false,
				Finished = false,
				ClearTextOnFocus = false,
				Placeholder = opts.Placeholder or "",
				Callback = function(v)
					text = v or ""
					if opts.Callback then
						pcall(opts.Callback, text)
					end
				end,
			})
		end)
		function state:get()
			return text
		end
		function state:set(v)
			text = v
			pcall(function()
				state._obj:SetValue(v)
			end)
		end
		return state
	end

	function api:keybind(opts)
		pcall(function()
			local label = group:AddLabel(opts.Text or "快捷键")
			label:AddKeyPicker(key("kp", opts), {
				Default = opts.Default and opts.Default.Name or "F",
				Mode = "Toggle",

				Callback = function(v)
					if opts.Callback then
						task.spawn(opts.Callback, v)
					end
				end,
			})
		end)
	end

	function api:colorpicker(opts)
		local state = {}
		local value = opts.Default or Color3.fromRGB(255, 255, 255)
		pcall(function()
			local label = group:AddLabel(opts.Text or "颜色")
			state._obj = label:AddColorPicker(key("cp", opts), {
				Default = value,
				Title = opts.Text or "颜色",
				Callback = function(v)
					value = v
					if opts.Callback then
						task.spawn(opts.Callback, v)
					end
				end,
			})
		end)
		function state:set(v)
			pcall(function()
				state._obj:SetValueRGB(v)
			end)
		end
		function state:get()
			return value
		end
		return state
	end

	function api:paragraph(text, height)
		local holder = { Text = text }
		local label
		pcall(function()
			label = group:AddLabel({ Text = text, DoesWrap = true })
		end)
		setmetatable(holder, {
			__newindex = function(t, k, v)
				rawset(t, k, v)
				if k == "Text" and label then
					pcall(function()
						label:SetText(v)
					end)
				end
			end,
		})
		return holder
	end

	function api:divider()
		pcall(function()
			group:AddDivider()
		end)
	end

	return api
end

function UIKit:createWindow(cfg)
	if not self._inited then
		self._inited = true
		local ok = pcall(function()
			self:init()
		end)
		if not ok then
			self.Backend = "native"
		end
	end

	if self.Backend == "obsidian" and self.Library then
		local ok, win = pcall(function()
			return self.Library:CreateWindow({
				Title = cfg.Title,
				Footer = "v" .. HUB.Version .. " · 地球",
				Icon = 0,
				NotifySide = "Right",
				Center = false,
				AutoShow = true,
			})
		end)
		if ok and win then
			local Library = self.Library
			HUB.Notify = function(text)
				pcall(function()
					Library:Notify({
						Title = "钻探至地球核心",
						Description = tostring(text),
						Time = 4,
					})
				end)
			end
			local wrapper = {}
			function wrapper:tab(name, order)
				local icon = TAB_ICONS[name] or "circle"
				local okTab, tab = pcall(function()
					return win:AddTab(name, icon)
				end)
				if okTab and tab then
					return wrapTab(tab)
				end
				return wrapTab({})
			end
			function wrapper:setOpen() end
			function wrapper:toggle() end
			return wrapper
		end
		self.Backend = "native"
	end

	local nativeWin = HUB.UI:createWindow(cfg)
	self.NativeWindow = nativeWin
	local wrapper = {}
	function wrapper:tab(name, order)
		return nativeWin:tab(name, order)
	end
	function wrapper:setOpen(v)
		nativeWin:setOpen(v)
	end
	function wrapper:toggle()
		nativeWin:toggle()
	end
	return wrapper
end

-- 物品 / 宝箱筛选列表自动更新（不用点刷新）
task.spawn(function()
	local lastCat, lastChests, lastPick = {}, "", {}
	while true do
		task.wait(3)
		pcall(function()
			local chests = HUB.listGroundChestNames(5000)
			local csig = table.concat(chests, "|")
			if csig ~= lastChests then
				lastChests = csig
				if #chests == 0 then
					chests = { "附近暂时没有宝箱" }
				end
				chestNameDd:refreshValues(chests)
			end
		end)
	end
end)

task.spawn(function()
	while true do
		task.wait(0.5)
		pcall(function()
			pcall(HUB.findPickTarget)
			local vals = S.Values or {}
			drawValueLabel.Text = "视野内卖钱物品总价值：" .. HUB.formatNum(vals.DrawTotal or 0)
			pickValueLabel.Text = "待捡卖钱物品总价值：" .. HUB.formatNum(vals.PickTotal or 0)
			local mined, picked = HUB.getMiningStats()
			mineStatLabel.Text = "已挖矿块：" .. tostring(mined) .. "　已拾取：" .. tostring(picked)
			local char = Players.LocalPlayer.Character
			local tool = char and char:FindFirstChildOfClass("Tool")
			weaponLabel.Text = "当前武器：" .. (tool and HUB.zhName(tool.Name) or "空手")
		end)
	end
end)

HUB.Notify("钻探至地球核心 v" .. HUB.Version .. " 已加载 · 点右侧圆钮打开", "ok", 5)
